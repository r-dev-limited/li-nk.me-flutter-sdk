cmake_minimum_required(VERSION 3.14)
set(PROJECT_NAME "flutter_linkme_sdk_windows")
project(${PROJECT_NAME} LANGUAGES CXX)

set(PLUGIN_NAME "flutter_linkme_sdk_windows_plugin")
add_library(${PLUGIN_NAME} SHARED "activation.cpp" "plugin.cpp")
apply_standard_settings(${PLUGIN_NAME})
target_compile_definitions(${PLUGIN_NAME} PRIVATE FLUTTER_PLUGIN_IMPL NOMINMAX)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter flutter_wrapper_plugin shell32)
set(flutter_linkme_sdk_windows_bundled_libraries "" PARENT_SCOPE)
